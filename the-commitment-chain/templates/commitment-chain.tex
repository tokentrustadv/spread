% commitment-chain.tex — custom Pandoc LaTeX template for
% "The Commitment Chain" (XeLaTeX only, requires fontspec).
%
% Structural notes for future maintainers:
%   - H1/H2 headings are NOT rendered via \part / \chapter. They are
%     intercepted by filters/cover-page.lua (PDF/LaTeX output only) and
%     turned into raw calls to \bookpart, \frontmatterchapter, or
%     \numberedchapter below, each of which registers its own TOC entry.
%     This is what lets the two true Parts ("Part I", "Part II") get a
%     visually distinct full-page treatment while ordinary front/back
%     matter (Copyright, Introduction, About the Author, ...) and the
%     numbered chapters share a simpler "chapter-opening" style — all
%     while the manuscript keeps a flat H1/H2/H3 hierarchy.
%   - H3 subsections are left as ordinary Pandoc headings and mapped to
%     \subsection via --top-level-division=chapter.
%   - The cover image is NOT included by this template. It is converted
%     to a one-page PDF and merged in front of this document's output by
%     the Makefile / GitHub Actions workflow, so the cover carries no
%     page number, header, or footer and never competes with this
%     document's own page-1 numbering.
\documentclass[$for(classoption)$$classoption$$sep$,$endfor$]{$documentclass$}

\usepackage{fontspec}
\setmainfont{TeX Gyre Pagella}
\setsansfont{TeX Gyre Heros}

\usepackage[$for(geometry)$$geometry$$sep$,$endfor$]{geometry}
\usepackage{xcolor}
\definecolor{accent}{HTML}{$accent-color$}
\definecolor{branddark}{HTML}{$brand-dark$}
\definecolor{accentdeep}{HTML}{3F5FBF}
\definecolor{bodytext}{HTML}{1A1A1A}

\usepackage{enumitem}
\setlist{itemsep=0.35em, topsep=0.5em, parsep=0pt}

\usepackage{microtype}
\usepackage{framed}
\usepackage{needspace}
\usepackage{etoolbox}
\usepackage{titlesec}
\usepackage{newunicodechar}
\newunicodechar{→}{\ensuremath{\rightarrow}}
\newunicodechar{←}{\ensuremath{\leftarrow}}

\usepackage[bookmarksnumbered=false,breaklinks=true]{hyperref}
\hypersetup{
  pdftitle={$title-meta$},
  pdfauthor={$author-meta$},
  pdfsubject={$subtitle$},
  pdfkeywords={$for(keywords)$$keywords$$sep$, $endfor$},
  colorlinks=true,
  linkcolor=accentdeep,
  citecolor=accentdeep,
  urlcolor=accentdeep,
  bookmarksopen=false,
}

% ---- Required Pandoc boilerplate ----
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

% ---- Typography ----
\AtBeginDocument{\color{bodytext}}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.65em}
\linespread{1.06}
\raggedbottom
\clubpenalty=10000
\widowpenalty=10000
\displaywidowpenalty=10000

% No LaTeX auto-numbering anywhere; our own macros supply chapter
% numerals, and headings otherwise carry no number.
\setcounter{secnumdepth}{-5}
% Only chapter-level (our custom \addcontentsline calls) entries show in
% the TOC; H3 subsections (real \subsection commands) are excluded.
\setcounter{tocdepth}{1}

% ---- Subsection (H3) styling: no forced page break, just a clear,
% modestly sized heading with breathing room, protected from orphaning
% at the bottom of a page. ----
\titleformat{\subsection}
  {\needspace{5\baselineskip}\normalfont\sffamily\bfseries\color{accentdeep}\Large}{}{0pt}{}
\titlespacing*{\subsection}{0pt}{1.3em}{0.6em}

% ---- Blockquotes: a restrained accent-colored left rule, italic text.
\renewcommand\FrameCommand{{\color{accent}\vrule width 2.6pt\relax}\hspace{10pt}}
\renewenvironment{quote}
  {\begin{leftbar}\itshape}
  {\end{leftbar}}

% ---- Running headers on ordinary body pages. Chapter/part-opening
% pages override this with \thispagestyle{plain} or {empty} below. ----
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\sffamily The Commitment Chain}
\fancyhead[R]{\small\itshape\nouppercase{\rightmark}}
\renewcommand{\headrulewidth}{0.5pt}
\renewcommand{\headrule}{{\color{accent}\hrule height\headrulewidth}}
\fancyfoot[C]{\small\thepage}
\renewcommand{\footrulewidth}{0pt}

% ---- Part dividers: full-page dark brand background, echoing the
% cover's palette, with no visible folio. ----
\newcommand{\bookpart}[2]{%
  \clearpage
  \phantomsection
  \addcontentsline{toc}{chapter}{#1: #2}
  \markboth{#2}{#2}
  \newgeometry{margin=0pt}
  \pagecolor{branddark}
  \thispagestyle{empty}
  \begin{center}
  \vspace*{0.32\paperheight}
  {\color{accent}\sffamily\bfseries\fontsize{18}{22}\selectfont \MakeUppercase{#1}}\\[0.7em]
  {\color{accent}\rule{0.16\paperwidth}{0.6pt}}\\[1.3em]
  {\color{white}\sffamily\bfseries\fontsize{38}{44}\selectfont #2}
  \end{center}
  \vfill\null
  \clearpage
  \restoregeometry
  \nopagecolor
}

% ---- Plain front/back-matter chapter openers (Copyright, Introduction,
% Conclusion, AI Reference, About the Author, Version Notes, and any
% other non-Part H1). ----
\newcommand{\frontmatterchapter}[2]{%
  \clearpage
  \thispagestyle{plain}
  \phantomsection
  \ifstrempty{#1}
    {\addcontentsline{toc}{chapter}{#2}\markboth{#2}{#2}}
    {\addcontentsline{toc}{chapter}{#1: #2}\markboth{#2}{#2}}
  \begingroup
  \raggedright
  \ifstrempty{#1}{}{%
    {\color{accentdeep}\sffamily\bfseries\fontsize{13}{15}\selectfont \MakeUppercase{#1}}\\[0.4em]
    {\color{accent}\rule{0.16\linewidth}{1.4pt}}\\[0.8em]}
  {\sffamily\bfseries\fontsize{27}{31}\selectfont #2\par}
  \endgroup
  \vspace{1.5em}
}

% ---- Numbered chapter openers (the eleven "N. Title" chapters). ----
\newcommand{\numberedchapter}[2]{%
  \clearpage
  \thispagestyle{plain}
  \phantomsection
  \addcontentsline{toc}{chapter}{#1. #2}
  \markboth{#2}{#2}
  \begingroup
  \raggedright
  {\color{accent}\sffamily\bfseries\fontsize{15}{15}\selectfont CHAPTER #1}\\[0.3em]
  {\color{accent}\rule{0.16\linewidth}{1.4pt}}\\[0.8em]
  {\sffamily\bfseries\fontsize{27}{31}\selectfont #2\par}
  \endgroup
  \vspace{1.5em}
}

\begin{document}
\pagenumbering{arabic}
$if(toc)$
\tableofcontents
\clearpage
$endif$
$body$
\end{document}
